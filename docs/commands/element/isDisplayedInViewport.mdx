# isDisplayedInViewport

## Overview {/* #overview */}

Use the `isDisplayedInViewport` command to determine whether a given DOM element is displayed within the viewport.

The command returns `true` if the selected DOM element is partially visible and within the viewport; otherwise, it returns `false`.

## Usage {/* #usage */}

```javascript
await browser.$(selector).isDisplayedInViewport();
```

## Usage Examples {/* #examples */}

**index.html**

```html testplane-html=element-html
<div id="visible">Visible</div>
<div id="notDisplayed" style="display: none">Hidden</div>
<div id="notVisible" style="visibility: hidden">Hidden</div>
<div id="notInViewport" style="position: absolute; left: 10000px">Off screen</div>
<div id="zeroOpacity" style="opacity: 0">Transparent</div>
```

**isDisplayedInViewport.js**

```javascript testplane=element-is-displayed-in-viewport fixture=element-html
const assert = require("node:assert/strict");

it("should detect if an element is visible", async ({ browser }) => {
    let elem = await browser.$("#visible");
    let isDisplayedInViewport = await elem.isDisplayedInViewport();
    console.log(isDisplayedInViewport); // outputs: true
    assert.equal(isDisplayedInViewport, true);

    elem = await browser.$("#notDisplayed");
    isDisplayedInViewport = await elem.isDisplayedInViewport();
    console.log(isDisplayedInViewport); // outputs: false
    assert.equal(isDisplayedInViewport, false);

    elem = await browser.$("#notVisible");
    isDisplayedInViewport = await elem.isDisplayedInViewport();
    console.log(isDisplayedInViewport); // outputs: false
    assert.equal(isDisplayedInViewport, false);

    elem = await browser.$("#notExisting");
    isDisplayedInViewport = await elem.isDisplayedInViewport();
    console.log(isDisplayedInViewport); // outputs: false
    assert.equal(isDisplayedInViewport, false);

    elem = await browser.$("#notInViewport");
    isDisplayedInViewport = await elem.isDisplayedInViewport();
    console.log(isDisplayedInViewport); // outputs: false
    assert.equal(isDisplayedInViewport, false);

    elem = await browser.$("#zeroOpacity");
    isDisplayedInViewport = await elem.isDisplayedInViewport();
    console.log(isDisplayedInViewport); // outputs: false
    assert.equal(isDisplayedInViewport, false);
});
```
